// Creating a Label and Input // We usually place a label next to an input field. // A label and input can also be connected so that clicking the label selects the corresponding input. const label = document.createElement("label"); label.textContent = "Name:"; label.htmlFor = "nameInput"; // Connect the label to the input with this id. const input = document.createElement("input"); input.type = "text"; input.id = "nameInput"; document.body.append(label, input);
