We create an array by placing items inside square brackets: const animals = ["cat", "dog", "monkey"]; The items remain in the order in which they were added. First: "cat" Second: "dog" Third: "monkey" An array will normally contain one type of information. Arrays preserve order, make it easy to add items, and allow us to work through an entire list using a loop. Lesson 1.2: Accessing and Changing Items INDEXES Each item in an array has a numbered position called an index. Array indexes begin at 0, not 1. const animals = ["cat", "dog", "monkey"]; const firstAnimal = animals[0]; // "cat" const secondAnimal = animals[1]; // "dog" const thirdAnimal = animals[2]; // "monkey" We can imagine the array like this: Index: 0 1 2 Item: "cat" "dog" "monkey" Starting at 0 may initially seem strange, but it is standard in TypeScript and many other programming languages. CHANGING AN ITEM We can replace an item by using its index: const animals = ["cat", "dog", "monkey"]; animals[1] = "tiger"; // Replace the second item. The array now contains: ["cat", "tiger", "monkey"] Even though animals was declared using const, we can change an item inside the array. The variable still refers to the same array.

Displaying Pokemon in List (v2)

Wouldn't it be nice if we could just write:

let pikachu = new Pokemon("Pikachu", "Electric", "Static", "Stores electricity in its cheeks.");

Use a little dark magic and we can.

class Pokemon { name: string; type: string; ability: string; funFact: string; constructor(name: string, type: string, ability: string, funFact: string) { // call let pikachu = new Pokemon("Pikachu", "Electric", "Static", "Stores electricity in its cheeks."); this.name = name; this.type = type; this.ability = ability; this.funFact = funFact; } } let myPokemon: Array = []; export function startApp() { createPokemon(); displayPokemon(); } function createPokemon() { let pikachu = new Pokemon("Pikachu", "Electric", "Static", "Stores electricity in its cheeks."); let charmander = new Pokemon("Charmander", "Fire", "Blaze", "The flame on its tail reflects its life force."); myPokemon.push(pikachu, charmander); } function displayPokemon() { for (const pokemon of myPokemon) { const text = pokemon.name + " " + pokemon.type //alert(text) } for (const pokemon of myPokemon) { pokemon.type = "N/A"; } for (const pokemon of myPokemon) { const text = pokemon.name + " " + pokemon.type //alert(text) } }
