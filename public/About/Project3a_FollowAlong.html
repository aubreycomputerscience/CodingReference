// GLOBAL VARIALBES const playerLabel = document.createElement("button"); // this is shared/permanent variables let xPosition = 100; let yPosition = 200; export function startApp() { // setup addPlayer() addRightButton(); addLeftButton(); } // BUILD UI function addPlayer() { playerLabel.textContent = "PLAYER"; playerLabel.style.fontSize = "40px"; playerLabel.style.position = "absolute"; updatePosition() // only update the position in one place // otherwise, you have to always update in 2 places // and it is easy to forget and can be hard to manage // in a large code base document.body.append(playerLabel); } function addLeftButton() { const leftButton = document.createElement("button"); leftButton.textContent = "<"; leftButton.style.fontSize = "40px"; leftButton.style.position = "absolute"; leftButton.style.top = "400px"; leftButton.style.left = "400px"; leftButton.onclick = () => { onLeftButtonPressed() }; document.body.append(leftButton); // the document stores this, so isn't deleted } function addRightButton() { const rightButton = document.createElement("button"); rightButton.textContent = ">"; rightButton.style.fontSize = "40px"; rightButton.style.position = "absolute"; rightButton.style.top = "400px"; rightButton.style.left = "450px"; rightButton.onclick = () => { onRightButtonPressed() }; document.body.append(rightButton); } // ON BUTTON PRESS function onRightButtonPressed() { xPosition = xPosition + 10; updatePosition() } function onLeftButtonPressed() { xPosition = xPosition - 10; updatePosition() } function updatePosition() { playerLabel.style.top = yPosition + "px"; playerLabel.style.left = xPosition + "px"; // TypeScript allows us to add a number and a string // This gives us a text string.
